"""T23 self-check for patch 0008 (RL loss): loss parity, RL-active grads,
KL-NaN failure branch, and the real anchor checkpoint load path.

Run: python3 /tmp/opencode/t23/selfcheck.py
"""
import importlib.util
import json
import logging
import os
import sys

import torch

logging.basicConfig(level=logging.INFO, format='%(levelname)s %(message)s')

ROOT = '/home/ygsldr/riichi-ai'
sys.path.insert(0, os.path.join(ROOT, 'vendor/mortal/mortal'))
sys.path.insert(0, os.path.join(ROOT, 'vendor/mortal/target/release'))

from libriichi.consts import ACTION_SPACE, obs_shape  # noqa: E402
import rl_loss  # noqa: E402
import model as model_new  # noqa: E402


def load_module(name, path):
    from importlib.machinery import SourceFileLoader
    loader = SourceFileLoader(name, path)
    spec = importlib.util.spec_from_file_location(name, path, loader=loader)
    mod = importlib.util.module_from_spec(spec)
    sys.modules[name] = mod
    spec.loader.exec_module(mod)
    return mod


model_base = load_module('model_base_t23', '/tmp/opencode/t23/model.py.base')

res = {
    'python': sys.version.split()[0],
    'torch': torch.__version__,
    'action_space': ACTION_SPACE,
    'obs_shape_v4': list(obs_shape(4)),
}
B = 16
torch.manual_seed(23)
dqn_old = model_base.DQN(version=4)
dqn_new = model_new.DQN(version=4)
dqn_new.load_state_dict(dqn_old.state_dict())

phi = torch.randn(B, 1024)
masks = torch.rand(B, ACTION_SPACE) > 0.25
masks[:, 0] = True
actions = torch.multinomial(masks.float(), 1).squeeze(1)
G = torch.randn(B).abs()
ranks = torch.randint(0, 4, (B,))
aux_logits = torch.randn(B, 4)

q_old = dqn_old(phi, masks)
q_new, v_new = dqn_new(phi, masks)
q_bitwise_equal = torch.equal(q_old, q_new)
v_expected_old = dqn_old.net(phi).split((1, ACTION_SPACE), dim=-1)[0]
v_bitwise_equal = torch.equal(v_new, v_expected_old)
res['forward_contract'] = {
    'q_bitwise_equal': q_bitwise_equal,
    'q_max_abs_diff_on_legal': float((q_old - q_new).masked_fill(~masks, 0.).detach().abs().max()),
    'v_bitwise_equal_to_old_head': v_bitwise_equal,
    'q_shape': list(q_new.shape),
    'v_shape': list(v_new.shape),
}
assert q_bitwise_equal and v_bitwise_equal

mse = torch.nn.MSELoss()
ce = torch.nn.CrossEntropyLoss()
min_q_weight, next_rank_weight = 5, 0.2


def old_loss(q_out):
    q = q_out[range(B), actions]
    dqn_loss = 0.5 * mse(q, G)
    cql_loss = q_out.logsumexp(-1).mean() - q.mean()
    next_rank_loss = ce(aux_logits, ranks)
    total = sum((dqn_loss, cql_loss * min_q_weight, next_rank_loss * next_rank_weight))
    return total


def new_loss(q_out, v, *, use_pg, lam_v, lam_kl, anchor_q=None):
    q = q_out[range(B), actions]
    dqn_loss = 0.5 * mse(q, G)
    cql_loss = q_out.logsumexp(-1).mean() - q.mean()
    next_rank_loss = ce(aux_logits, ranks)
    loss = sum((dqn_loss, cql_loss * min_q_weight, next_rank_loss * next_rank_weight))
    pg, vl, kl = rl_loss.rl_loss_terms(
        q_out = q_out, v = v, actions = actions, returns = G, masks = masks,
        use_pg = use_pg, lam_v = lam_v, lam_kl = lam_kl,
        anchor_q = anchor_q, anchor_desc = 'SELFTEST',
    )
    return loss + pg + vl + kl, pg, vl, kl


loss_old = old_loss(q_old)
loss_new0, pg0, vl0, kl0 = new_loss(q_new, v_new, use_pg=False, lam_v=1.0, lam_kl=0.0)
res['parity_use_pg_false_lam_kl_0'] = {
    'old': float(loss_old),
    'new': float(loss_new0),
    'delta': float(loss_new0 - loss_old),
    'old_hex': float(loss_old).hex(),
    'new_hex': float(loss_new0).hex(),
    'bit_identical': float(loss_new0) == float(loss_old),
    'rl_terms_exactly_zero': float(pg0) == 0.0 and float(vl0) == 0.0 and float(kl0) == 0.0,
    'below_1e-6': abs(float(loss_new0 - loss_old)) < 1e-6,
}
assert float(loss_new0 - loss_old) == 0.0
assert res['parity_use_pg_false_lam_kl_0']['rl_terms_exactly_zero']

loss_new1, *_ = new_loss(q_new, v_new, use_pg=False, lam_v=1.0, lam_kl=0.05)
res['parity_use_pg_false_lam_kl_005_master_switch'] = {
    'delta': float(loss_new1 - loss_old),
    'bit_identical': float(loss_new1) == float(loss_old),
}
assert float(loss_new1 - loss_old) == 0.0

phi_g = torch.randn(B, 1024, requires_grad=True)
q_g, v_g = dqn_new(phi_g, masks)
anchor_q = torch.randn(B, ACTION_SPACE)
anchor_q = anchor_q.masked_fill(~masks, -torch.inf)
loss_g, pg_g, vl_g, kl_g = new_loss(q_g, v_g, use_pg=True, lam_v=1.0, lam_kl=0.05, anchor_q=anchor_q)
res['rl_active'] = {
    'pg': float(pg_g),
    'huber_v': float(vl_g),
    'kl_scaled': float(kl_g),
    'all_finite': all(bool(torch.isfinite(t)) for t in (pg_g, vl_g, kl_g)),
    'nonzero': float(pg_g) != 0.0 and float(vl_g) != 0.0 and float(kl_g) != 0.0,
}
assert res['rl_active']['all_finite'] and res['rl_active']['nonzero']
loss_g.backward()
res['rl_active']['phi_grad_finite'] = bool(torch.isfinite(phi_g.grad).all())
res['rl_active']['phi_grad_l1'] = float(phi_g.grad.abs().sum())
res['rl_active']['dqn_a_head_grad_norm'] = float(dqn_new.a_head.weight.grad.norm()) if hasattr(dqn_new, 'a_head') else float(dqn_new.net.weight.grad.norm())
assert res['rl_active']['phi_grad_finite'] and res['rl_active']['phi_grad_l1'] > 0

bad_anchor = anchor_q.clone()
bad_anchor[0, actions[0]] = float('nan')
try:
    rl_loss.rl_loss_terms(
        q_out=q_g, v=v_g, actions=actions, returns=G, masks=masks,
        use_pg=True, lam_v=1.0, lam_kl=0.05,
        anchor_q=bad_anchor, anchor_desc='SELFTEST_NAN_ANCHOR',
    )
    res['kl_nan_branch'] = {'exit_code': None, 'raised': False}
except SystemExit as ex:
    res['kl_nan_branch'] = {'exit_code': ex.code, 'raised': True}
assert res['kl_nan_branch']['raised'] and res['kl_nan_branch']['exit_code'] == 1

ckpt = os.path.join(ROOT, 'checkpoints/mortal.pth.at-148800')
state = torch.load(ckpt, weights_only=True, map_location='cpu')
cfg = state['config']
anchor_version = cfg['control'].get('version', 1)
anchor_mortal = model_new.brain_from_config(cfg).eval()
anchor_dqn = model_new.DQN(version=anchor_version).eval()
model_new.load_brain_state_dict(anchor_mortal, state['mortal'], source=ckpt)
anchor_dqn.load_state_dict(state['current_dqn'])
obs = torch.zeros(2, *obs_shape(anchor_version))
legal = torch.ones(2, ACTION_SPACE, dtype=torch.bool)
with torch.no_grad():
    anchor_phi = anchor_mortal(obs, events=None)
    anchor_q_real, anchor_v_real = anchor_dqn(anchor_phi, legal)
res['real_anchor_load'] = {
    'path': ckpt,
    'version': anchor_version,
    'steps': state.get('steps'),
    'config_has_model_section': 'model' in cfg,
    'q_shape': list(anchor_q_real.shape),
    'v_shape': list(anchor_v_real.shape),
    'all_finite': bool(torch.isfinite(anchor_q_real).all() and torch.isfinite(anchor_v_real).all()),
    'kl_vs_live_finite': bool(torch.isfinite(rl_loss.forward_kl_loss(anchor_q_real, q_new[:2], masks[:2], anchor_desc='SELFTEST_REAL_ANCHOR'))),
}
assert res['real_anchor_load']['all_finite'] and res['real_anchor_load']['kl_vs_live_finite']

out = '/tmp/opencode/t23/selfcheck.json'
with open(out, 'w') as f:
    json.dump(res, f, indent=2)
print('SELFCHECK_JSON', out)
print(json.dumps(res, indent=2))
print('SELFCHECK_PASS')
